class Button:

    def __init__(self, x, y, w, h, text, command):

        self.rect = pygame.Rect(x, y, w, h)

        self.text = text

        self.command = command

    def draw(self):

        pygame.draw.rect(
            screen,
            DARK_GREEN,
            self.rect,
            border_radius=10
        )

        pygame.draw.rect(
            screen,
            GREEN,
            self.rect,
            2,
            border_radius=10
        )

        txt = font.render(self.text, True, WHITE)

        screen.blit(
            txt,
            (
                self.rect.x + 20,
                self.rect.y + 12
            )
        )

    def clicked(self, pos):

        return self.rect.collidepoint(pos)

    screen.fill(BLACK)

        try:

            current_distance = data["distance"]

            flame_detected = data["flame"]

            gate_closed = data["gate_closed"]

            ),
            math.pi,
            2 * math.pi,

        4

    # RADAR BLIPS

    for angle, distance in blips:

        if distance > MAX_DISTANCE:

        rad = math.radians(angle)

        radius = (distance / MAX_DISTANCE) * 500

        x = CENTER_X + radius * math.cos(rad)
        y = CENTER_Y - radius * math.sin(rad)

        pygame.draw.circle(
            screen,
            RED,
            (int(x), int(y)),
            5
        )

    # =====================================================
    # STATUS PANEL
    # =====================================================

    status_x = 920

    title = font.render("SMART HOME STATUS", True, GREEN)

        WHITE

    # DRAW BUTTONS
    # =====================================================

    for button in buttons:

        button.draw()

    # =====================================================
    # EVENTS
    # =====================================================

# Ultrasonic
TRIG_PIN = 5
ECHO_PIN = 27

# =========================================================
# LED SETUP

leds = []

for pin in LED_PINS:

    leds.append(
        Pin(pin, Pin.OUT)
    )

# =========================================================
# BUZZER

buzzer = Pin(BUZZER_PIN, Pin.OUT)

# =========================================================
# FLAME + IR

flame_sensor = Pin(FLAME_PIN, Pin.IN)

ir_sensor = Pin(IR_PIN, Pin.IN)

# =========================================================
# SERVO
# =========================================================

servo = PWM(
    Pin(SERVO_PIN),
    freq=50
)

MIN_ANGLE = 15
MAX_ANGLE = 165

STEP = 3

current_angle = MIN_ANGLE

direction = STEP

def set_angle(angle):

    if angle < MIN_ANGLE:
        angle = MIN_ANGLE

    if angle > MAX_ANGLE:
        angle = MAX_ANGLE

    duty = int((angle / 180) * 75 + 40)

    servo.duty(duty)

# =========================================================
# ULTRASONIC
# =========================================================

trig = Pin(TRIG_PIN, Pin.OUT)

echo = Pin(ECHO_PIN, Pin.IN)

def get_distance():

    trig.off()

    time.sleep_us(2)

    trig.on()

    time.sleep_us(10)

    trig.off()

    try:

        duration = time_pulse_us(
            echo,
            1,
            30000
        )

        if duration < 0:
            return None

        distance = (
            duration * 0.0343
        ) / 2

        if distance <= 0 or distance > 400:
            return None

        return round(distance, 2)

    except:

        return None

# =========================================================
# LED CONTROL
# =========================================================

def set_room(room, state):

    if 1 <= room <= 4:

        leds[room - 1].value(state)

def set_all(state):

    for led in leds:

        led.value(state)

# =========================================================
# BUZZER FUNCTIONS
# =========================================================

def short_beep():

    buzzer.on()

    time.sleep(0.1)

    buzzer.off()

# =========================================================
# COMMAND HANDLER
# =========================================================

def handle_command(cmd):

    cmd = cmd.strip().upper()

    # ---------------------------
    # ROOM CONTROL
    # ---------------------------

    if cmd == "R1 ON":
        set_room(1, 1)

    elif cmd == "R1 OFF":
        set_room(1, 0)

    elif cmd == "R2 ON":
        set_room(2, 1)

    elif cmd == "R2 OFF":
        set_room(2, 0)

    elif cmd == "R3 ON":
        set_room(3, 1)

    elif cmd == "R3 OFF":
        set_room(3, 0)

    elif cmd == "R4 ON":
        set_room(4, 1)

    elif cmd == "R4 OFF":
        set_room(4, 0)

    elif cmd == "ALL ON":
        set_all(1)

    elif cmd == "ALL OFF":
        set_all(0)

    # ---------------------------
    # BUZZER
    # ---------------------------

    elif cmd == "BEEP":

        short_beep()

# NON-BLOCKING SERIAL INPUT
# =========================================================

poller = uselect.poll()

poller.register(

# STARTUP

set_all(0)

import network
import time
import ujson

from hardware import *

CONFIG_FILE = "config.json"

# =====================================
# LOAD CONFIG
# =====================================

def load_config():

    try:

        with open(CONFIG_FILE, "r") as f:

            return ujson.load(f)

    except:

        return None

# =====================================
# SAVE CONFIG
# =====================================

def save_config(data):

    with open(CONFIG_FILE, "w") as f:

        ujson.dump(data, f)

# =====================================
# CONNECT WIFI
# =====================================

def connect_wifi():

    config = load_config()

    if not config:

        return None

    ssid = config["wifi_ssid"]

    password = config["wifi_password"]

    wlan = network.WLAN(network.STA_IF)

    wlan.active(True)

    wlan.connect(ssid, password)

    print("Connecting to WiFi...")

    timeout = 20

    while timeout > 0:

        if wlan.isconnected():

            print("WiFi Connected")

            print("IP:", wlan.ifconfig()[0])

            wifi_connected()

            return wlan

        wifi_blink()

        timeout -= 1

    print("WiFi Failed")

    return None

    ap = network.WLAN(network.AP_IF)

    ap.active(True)

    ap.config(
        essid=AP_NAME,
        password=AP_PASSWORD
    )

    print("Access Point Started")
    print("Connect to:", AP_NAME)

    addr = socket.getaddrinfo(
        "0.0.0.0",
        80
    )[0][-1]

    s = socket.socket()

    s.bind(addr)

    s.listen(1)

    print("Web server running")

    while True:

        client, addr = s.accept()

        print("Client connected")

        request = client.recv(1024).decode()

        # =================================
        # SAVE CONFIG
        # =================================

        if "/save?" in request:

            try:

                ssid = ure.search(
                    "ssid=([^&]*)",
                    request
                ).group(1)

                password = ure.search(
                    "password=([^&]*)",
                    request
                ).group(1)

                api = ure.search(
                    "api=([^ ]*)",
                    request
                ).group(1)

                data = {

                    "wifi_ssid": ssid,

                    "wifi_password": password,

                    "api_base": api
                }

                save_config(data)

                response = """
                <html>
                <body style='font-family:sans-serif'>
                <h2>Saved Successfully</h2>
                <p>ESP32 restarting...</p>
                </body>
                </html>
                """

                client.send(response)

                client.close()

                machine.reset()

            except Exception as e:

                print("Save Error:", e)

        # =================================
        # CONFIG PAGE
        # =================================

        else:

            html = """
            <html>
            <head>
                <title>Smart Home Setup</title>
            </head>

            <body style='font-family:sans-serif;padding:20px'>

                <h2>ESP32 WiFi Setup</h2>

                <form action="/save">

                    <p>WiFi Name</p>
                    <input name="ssid">

                    <p>Password</p>
                    <input name="password" type="password">

                    <p>API Base URL</p>
                    <input name="api">

                    <br><br>

                    <button type="submit">
                        Save
                    </button>

                </form>

            </body>
            </html>
            """

            client.send(html)
